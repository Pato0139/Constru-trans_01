    %% ==============================
    %% ENTIDADES PRINCIPALES
    %% ==============================

    %% Usuarios y Perfiles
    Usuario {
        int id PK
        string username
        string password
        string email
        string nombres
        string apellidos
        string documento
        string tipo_documento
        string rol
        string estado
        boolean sincronizado
    }

    Cliente {
        int id PK
        int usuario_id FK
        string nombre_empresa
        string direccion_principal
        string tipo_cliente
    }

    EPS {
        int id PK
        string codigo_eps
        string numero_seguro
        string ciudad
        string direccion
        string telefono
        string correo
    }

    Conductor {
        int id PK
        int usuario_id FK
        string numero_licencia
        string categoria_licencia
        date fecha_vencimiento_licencia
        string estado
        int eps_id FK
    }

    %% Vehículos y Asignaciones
    Vehiculo {
        int id PK
        string placa
        string marca
        string modelo
        string tipo_vehiculo
        float capacidad_carga
        string estado
    }

    ConductorVehiculo {
        int id PK
        int conductor_id FK
        int vehiculo_id FK
    }

    %% Inventario y Materiales
    Proveedor {
        int id PK
        string nit
        string nombre_empresa
        string telefono
        string correo
        string descripcion
    }

    MaterialConstruccion {
        int id PK
        string nombre
        string unidad_medida
        decimal precio
        string descripcion
    }

    Stock {
        int id PK
        int material_id FK
        int cantidad_actual
        int stock_minimo
        string ubicacion
    }

    %% Ventas y Pedidos
    MetodoPago {
        int id PK
        string codigo_metodo_pago
        string metodo
    }

    Pedido {
        int id PK
        int usuario_id FK
        int cliente_id FK
        string codigo_pedido
        string direccion_origen
        string direccion_destino
        decimal precio
        decimal total
        string estado
        datetime fecha_solicitud
        datetime fecha_entrega_programada
    }

    DetallePedido {
        int id PK
        int pedido_id FK
        int material_id FK
        int cantidad
        decimal precio_unitario
    }

    Entrega {
        int id PK
        int pedido_id FK
        int conductor_id FK
        int vehiculo_id FK
        string direccion_entrega
        datetime fecha_salida
        datetime fecha_entrega
        string estado
    }

    %% Facturación y Pagos
    Factura {
        int id_factura PK
        int pedido_id FK
        int cliente_id FK
        string numero
        datetime fecha
        decimal subtotal
        decimal iva
        decimal total
        string estado
    }

    Pago {
        int id_pago PK
        int factura_id FK
        decimal monto
        datetime fecha
        int codigo_metodo_pago_id FK
        string referencia
        int registrado_por_id FK
    }

    %% Compras
    Compra {
        int id_compra PK
        int proveedor_id FK
        datetime fecha_compra
        decimal total_compra
        string estado
        int usuario_id FK
    }

    DetalleCompra {
        int id_detalle_compra PK
        int compra_id FK
        int material_id FK
        int cantidad
        decimal precio_unitario
    }

    %% Movimientos y Auditoría
    MovimientoInventario {
        int id PK
        int material_id FK
        string tipo_movimiento
        int cantidad
        datetime fecha_movimiento
        int usuario_id FK
    }

    Notificacion {
        int id PK
        int usuario_id FK
        string titulo
        string mensaje
        boolean leida
        datetime fecha
    }

    Historial {
        int id PK
        int usuario_id FK
        string accion
        string tabla
        int registro_id
        datetime fecha
        string ip
    }

    %% Reportes
    Reporte {
        int id PK
        string titulo
        string tipo
        datetime fecha_creacion
    }

    HistorialReporte {
        int id PK
        int reporte_id FK
        datetime fecha_exportacion
    }

    %% ==============================
    %% RELACIONES
    %% ==============================

    %% Relaciones de Usuario
    Usuario ||--o| Cliente : "perfil_cliente"
    Usuario ||--o| Conductor : "perfil_conductor"
    Usuario ||--o{ Notificacion : "notificaciones"
    Usuario ||--o{ Pedido : "pedidos"
    Usuario ||--o{ MovimientoInventario : "movimientos"
    Usuario ||--o{ Compra : "compras"
    Usuario ||--o{ Pago : "pagos_registrados"
    Usuario ||--o{ Historial : "auditoria"

    %% Relaciones de EPS y Conductor
    EPS ||--o{ Conductor : "conductores"

    %% Relaciones de Vehículos
    Conductor ||--o{ ConductorVehiculo : "vehiculos_asignados"
    Vehiculo ||--o{ ConductorVehiculo : "conductores_asignados"

    %% Relaciones de Proveedores y Compras
    Proveedor ||--o{ Compra : "ventas"
    Compra ||--|{ DetalleCompra : "detalles"
    MaterialConstruccion ||--o{ DetalleCompra : "compras"

    %% Relaciones de Inventario
    MaterialConstruccion ||--o| Stock : "stock_info"
    MaterialConstruccion ||--o{ DetallePedido : "detalles_pedidos"
    MaterialConstruccion ||--o{ MovimientoInventario : "movimientos"

    %% Relaciones de Pedidos
    Pedido ||--|{ DetallePedido : "detalles"
    Pedido ||--o| Factura : "factura"
    Pedido ||--o| Entrega : "entrega"
    Pedido ||--o{ Pago : "pagos"

    %% Relaciones de Facturación y Pagos
    Factura ||--o{ Pago : "pagos"
    MetodoPago ||--o{ Pago : "pagos"

    %% Relaciones de Entregas
    Conductor ||--o{ Entrega : "entregas"
    Vehiculo ||--o{ Entrega : "entregas"

    %% Relaciones de Reportes
    Reporte ||--o{ HistorialReporte : "exportaciones"
